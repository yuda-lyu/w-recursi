/**
 * cfg.governance 之形檢查(純函式;createRsi 據以拋錯):須為物件、無未知鍵、requireReview 為布林、maxPendingRounds 為非負整數、gateKinds 為類別陣列;各鍵為 undefined 或 null 時取預設
 *
 * @param {*} gc 輸入 cfg.governance
 * @returns {Object} 回傳 { errors:Array(字串) }
 * @example
 * console.log(checkGovernanceCfg(undefined).errors.length, checkGovernanceCfg({ requireReview: 1 }).errors.length, checkGovernanceCfg({ foo: 1 }).errors.length, checkGovernanceCfg({ gateKinds: null }).errors.length)
 * // => 0 1 1 0
 */
export let checkGovernanceCfg = (gc) => {
    let errors = []
    if (gc === undefined || gc === null) { return { errors } }
    if (!isPlainObject(gc)) { return { errors: ['cfg.governance 須為物件 { requireReview?, maxPendingRounds?, gateKinds? }'] } }
    let unknown = keys(gc).filter((k) => !GOVERNANCE_CFG_KEYS.includes(k))
    if (unknown.length) { errors.push(`cfg.governance 有未知鍵 ${unknown.join(', ')}(已知鍵 ${GOVERNANCE_CFG_KEYS.join(', ')})`) }
    if (!isNil(gc.requireReview) && !isBoolean(gc.requireReview)) { errors.push('cfg.governance.requireReview 須為布林') }
    if (!isNil(gc.maxPendingRounds) && !(Number.isSafeInteger(gc.maxPendingRounds) && gc.maxPendingRounds >= 0)) { errors.push('cfg.governance.maxPendingRounds 須為非負整數') }
    if (!isNil(gc.gateKinds) && !(isArray(gc.gateKinds) && gc.gateKinds.every((k) => PROPOSAL_KINDS.includes(k)))) { errors.push(`cfg.governance.gateKinds 須為類別陣列(${PROPOSAL_KINDS.join(' / ')})`) }
    return { errors }
}

/**
 * 提案之檢查(純函式):須為陣列;每筆之 kind 在值域、mechanism 非空(機制之中性描述;revise = 修訂後之定義;retire = 停用之理由;note = 事項之內容)、new / layer / note 須附 title、revise / retire 須附 arm(既有臂之 id)、其餘字串欄須為字串、sources 須為字串陣列
 *
 * @param {*} items 輸入提案陣列 [{ kind, arm?, layer?, title?, mechanism, boundary?, evidence?, sources? }]
 * @returns {Object} 回傳 { errors:Array(字串) }
 * @example
 * let ok = [{ kind: 'new', layer: 'P', title: 'z 步進', mechanism: '對 z 軸步進' }, { kind: 'retire', arm: 'tag', mechanism: '與 px 重疊' }]
 * console.log(checkProposals(ok).errors.length, checkProposals([{ kind: 'new', mechanism: 'm' }, { kind: 'merge' }]).errors.length, checkProposals('x').errors.length)
 * // => 0 2 1
 */
export let checkProposals = (items) => {
    if (!isArray(items)) { return { errors: ['提案須為陣列'] } }
    let errors = []
    items.forEach((x, i) => {
        let at = `提案[${i}]`
        if (!isPlainObject(x) || !PROPOSAL_KINDS.includes(x.kind)) { errors.push(`${at}:kind 須為 ${PROPOSAL_KINDS.join(' | ')}`); return }
        if (!str(x.mechanism)) { errors.push(`${at}:mechanism 不得為空(機制之中性描述;revise = 修訂後之定義;retire = 停用之理由;note = 事項之內容)`); return }
        if (['new', 'layer', 'note'].includes(x.kind) && !str(x.title)) { errors.push(`${at}:kind ${x.kind} 須附 title`); return }
        if (['revise', 'retire'].includes(x.kind) && !str(x.arm)) { errors.push(`${at}:kind ${x.kind} 須附 arm(既有臂之 id)`); return }
        if (['arm', 'layer', 'title', 'boundary', 'evidence'].some((k) => !isNil(x[k]) && !isString(x[k])) || (!isNil(x.sources) && !(isArray(x.sources) && x.sources.every(isString)))) { errors.push(`${at}:須為 { kind, arm?, layer?, title?, mechanism, boundary?, evidence?, sources?[] }(字串欄須為字串)`) }
    })
    return { errors }
}

/**
 * 把擴充套到目錄上(純函式;不改入參):擴充 = { layers:[{ id, title }], arms:[{ op:"add", arm:{…} } | { op:"revise", id, patch:{…} } | { op:"retire", id, why }] },依序套用;各筆之 origin(出處)寫入該臂之 origin 欄(修訂記於 origin.revised、停用記於 origin.retired)。新臂之 n 接全目錄之最大序號、status 未給為 proposed;修訂之 patch 中值為 null 之鍵 = 刪除該鍵(必填欄 title / status / realize 與有層表時之 layer 不得刪;刪後之目錄仍須合格,如 blocked 之臂之 blockedBy 由結構檢查擋);停用者改為 blocked、不可選並留在目錄。擴充為空 ⇒ 原樣回傳目錄;結構不合(層或臂重複、層不存在、基底無層表而新增層、要改之臂不存在、修訂之欄不可改或未知、刪除必填欄、op 不明)⇒ 拋錯
 *
 * @param {Array|Object|null} catalog 輸入基底目錄(臂陣列或 { arms, layers? })
 * @param {Object|null} ext 輸入擴充
 * @returns {Array|Object|null} 回傳套用後之目錄(有擴充時為 { …原欄, layers, arms } 物件形)
 * @throws {Error} 擴充之結構不合時拋出
 * @example
 * let base = { layers: { P: '位置層' }, arms: [{ id: 'px', n: 1, layer: 'P', title: 'x 步進', status: 'sparse', realize: [{ kind: 'operator', op: 'P.x' }] }] }
 * let ext = {
 *     layers: [{ id: 'K', title: '資訊源層' }],
 *     arms: [
 *         { op: 'add', arm: { id: 'kz', layer: 'K', title: 'z 來源', realize: [{ kind: 'manual', how: '待實作' }] }, origin: { proposal: 'P001' } },
 *         { op: 'retire', id: 'px', why: '併入 kz' },
 *     ],
 * }
 * let out = applyArmExtensions(base, ext)
 * console.log(Object.keys(out.layers).join(','), out.arms.map((a) => `${a.id}:${a.status}:${a.n}`).join(' '), out.arms[1].origin.proposal, base.arms.length)
 * // => P,K px:blocked:1 kz:proposed:2 P001 1
 *
 * console.log(applyArmExtensions(base, null) === base)
 * // => true
 *
 * let base2 = { arms: [{ id: 'py', title: 'y 步進', status: 'sparse', grid: { y: [1, 2] }, realize: [{ kind: 'manual', how: '—' }] }] }
 * console.log('grid' in applyArmExtensions(base2, { arms: [{ op: 'revise', id: 'py', patch: { grid: null } }] }).arms[0])
 * // => false
 */
export let applyArmExtensions = (catalog, ext) => {
    let xl = isArray(get(ext, 'layers')) ? ext.layers : []
    let xa = isArray(get(ext, 'arms')) ? ext.arms : []
    if (!xl.length && !xa.length) { return catalog }
    let baseLayers = catalogLayersOf(catalog)
    let layers = { ...baseLayers }
    let arms = catalogArmsOf(catalog).map((a) => ({ ...a }))
    if (xl.length && !keys(baseLayers).length) { throw new Error('臂目錄之擴充:基底目錄無層表(catalog.layers), 不能新增層') }
    for (let [i, l] of xl.entries()) {
        if (!isPlainObject(l) || !isString(l.id) || !ID_RE.test(l.id) || !str(l.title)) { throw new Error(`臂目錄之擴充:layers[${i}] 須為 { id(識別字), title }`) }
        if (layers[l.id] !== undefined) { throw new Error(`臂目錄之擴充:層 ${l.id} 已存在`) }
        layers[l.id] = str(l.title)
    }
    let hasLayers = keys(layers).length > 0
    let maxN = () => arms.reduce((m, x) => Math.max(m, Number.isInteger(x.n) ? x.n : 0), 0)
    for (let [i, e] of xa.entries()) {
        if (!isPlainObject(e)) { throw new Error(`臂目錄之擴充:arms[${i}] 須為物件`) }
        let origin = isPlainObject(e.origin) ? e.origin : null
        if (e.op === 'add') {
            //值為 undefined 之鍵視同未給(與 decide 一致;2026-11-14 外審 Argon #5)
            let a = isPlainObject(e.arm) ? omitBy(e.arm, isUndefined) : e.arm
            if (!isPlainObject(a) || !isString(a.id)) { throw new Error(`臂目錄之擴充:arms[${i}] 之 add 須附 arm(含 id)`) }
            if (arms.some((x) => x.id === a.id)) { throw new Error(`臂目錄之擴充:臂 ${a.id} 已存在`) }
            if (hasLayers && !(a.layer in layers)) { throw new Error(`臂目錄之擴充:臂 ${a.id} 之層 ${a.layer} 不存在(先提新層)`) }
            let useN = arms.some((x) => Number.isInteger(x.n))
            arms.push({ ...(useN ? { n: maxN() + 1 } : {}), status: 'proposed', ...omit(a, ['n', 'origin']), ...(origin ? { origin } : {}) })
        }
        else if (e.op === 'revise') {
            let k = arms.findIndex((x) => x.id === e.id)
            if (k < 0) { throw new Error(`臂目錄之擴充:要修訂之臂 ${e.id} 不存在`) }
            let p = isPlainObject(e.patch) ? omitBy(e.patch, isUndefined) : {}
            let bad = keys(p).filter((x) => !REVISABLE.includes(x))
            if (bad.length) { throw new Error(`臂目錄之擴充:修訂 ${e.id} 之欄 ${bad.join(', ')} 不可改或未知(可改之欄 ${REVISABLE.join(', ')})`) }
            if (!keys(p).length) { throw new Error(`臂目錄之擴充:修訂 ${e.id} 須附 patch(可改之欄 ${REVISABLE.join(', ')})`) }
            //值為 null 之鍵 = 刪除該鍵(2026-11-14,Gitea #144);必填欄不得刪
            let del = keys(p).filter((x) => p[x] === null)
            let req = del.filter((x) => REQUIRED_KEYS.includes(x) || (x === 'layer' && hasLayers))
            if (req.length) { throw new Error(`臂目錄之擴充:修訂 ${e.id} 之必填欄 ${req.join(', ')} 不得刪除(null)`) }
            if (hasLayers && !isNil(p.layer) && !(p.layer in layers)) { throw new Error(`臂目錄之擴充:臂 ${e.id} 之層 ${p.layer} 不存在`) }
            arms[k] = { ...omit(arms[k], del), ...omit(p, del), ...(origin ? { origin: { ...(arms[k].origin || {}), revised: origin } } : {}) }
        }
        else if (e.op === 'retire') {
            let k = arms.findIndex((x) => x.id === e.id)
            if (k < 0) { throw new Error(`臂目錄之擴充:要停用之臂 ${e.id} 不存在`) }
            arms[k] = { ...omit(arms[k], ['selectable', 'requires']), status: 'blocked', blockedBy: `停用(治理):${str(e.why) || '未註明'}`, ...(origin ? { origin: { ...(arms[k].origin || {}), retired: origin } } : {}) }
        }
        else { throw new Error(`臂目錄之擴充:arms[${i}] 之 op 須為 add / revise / retire`) }
    }
    return { ...(isPlainObject(catalog) ? catalog : {}), layers, arms }
}

/**
 * 讀狀態根之擴充檔(<root>/governance/extensions.json):不存在 ⇒ null;存在而非 JSON 或形不合(須為 { layers?:[], arms?:[] })⇒ 拋錯(靜默略過會使已審定之臂無聲消失)
 *
 * @param {Object} paths 輸入 makePaths 之回傳(取 govExtensions)
 * @returns {Object|null} 回傳 { layers:Array, arms:Array } 或 null
 * @throws {Error} 擴充檔非 JSON 或形不合時拋出
 * @example
 * import fs from 'fs'
 * import os from 'os'
 * import path from 'path'
 * import { makePaths } from 'w-recursi/src/core/paths.mjs'
 *
 * let dir = fs.mkdtempSync(path.join(os.tmpdir(), 'w-recursi-'))
 * let paths = makePaths(dir)
 * console.log(readArmExtensions(paths))
 * // => null
 * fs.mkdirSync(paths.governance)
 * fs.writeFileSync(paths.govExtensions, '{bad')
 * try { readArmExtensions(paths) } catch (e) { console.log(e.message.startsWith('臂目錄之擴充:擴充檔非 JSON')) }
 * // => true
 * fs.rmSync(dir, { recursive: true, force: true })
 */
export let readArmExtensions = (paths) => {
    let fp = paths.govExtensions
    if (!fs.existsSync(fp)) { return null }
    let e = null
    try { e = readJson(fp, null) }
    catch (err) { throw new Error(`臂目錄之擴充:擴充檔非 JSON(${fp})`) }
    if (!isPlainObject(e) || (!isNil(e.layers) && !isArray(e.layers)) || (!isNil(e.arms) && !isArray(e.arms))) { throw new Error(`臂目錄之擴充:擴充檔須為 { layers?:[], arms?:[] }(${fp})`) }
    return { layers: isArray(e.layers) ? e.layers : [], arms: isArray(e.arms) ? e.arms : [] }
}

/**
 * 擴充之臂中,落成名(realize)之登錄與否可能隨擴充而變者之 id(供 makeArms / validateCatalog 之 lenient):add 之臂、retire 之臂、patch 含 realize / selectable / status(含以 null 刪除)之 revise 之臂 —— 由「可選臂」推導算子之領域,停用或改變可選與否後之下一次組裝即不再登錄其算子(2026-11-14,Gitea #144);只改其他欄之修訂不列入,基底之落成名仍嚴格
 *
 * @param {Object|null} ext 輸入擴充 { layers, arms }
 * @returns {Array|null} 回傳臂 id 陣列;無擴充 ⇒ null
 * @example
 * let ext = { arms: [{ op: 'add', arm: { id: 'kz' } }, { op: 'revise', id: 'px', patch: { title: 't' } }, { op: 'revise', id: 'py', patch: { realize: [] } }, { op: 'revise', id: 'pq', patch: { selectable: false } }, { op: 'retire', id: 'pair' }] }
 * console.log(lenientIdsOf(ext).join(','), lenientIdsOf(null))
 * // => kz,py,pq,pair null
 */
export let lenientIdsOf = (ext) => {
    if (!ext) { return null }
    let soft = (e) => isPlainObject(e) && ((e.op === 'add' && isPlainObject(e.arm)) || (e.op === 'revise' && isPlainObject(e.patch) && LENIENT_PATCH_KEYS.some((k) => e.patch[k] !== undefined)) || (e.op === 'retire' && isString(e.id)))
    return (isArray(ext.arms) ? ext.arms : []).filter(soft).map((e) => (e.op === 'add' ? e.arm.id : e.id))
}

/**
 * 擴充所觸及之臂(lenientIdsOf)之落成名(realize 之 op / mech / module / name 等字串欄),取自基底目錄之該臂與擴充項(add 之 arm.realize、revise 之 patch.realize)——供 makeArms / validateCatalog 之 lenientNames:由目錄推導登錄者,這些名之登錄與否隨審定而變(停用或改落成名後不再登錄),共用其名之他臂(含基底未被觸及者)亦不因之使組裝拋錯(2026-11-14,Gitea #144);套用後之目錄內 lenient 之臂之名由 validateCatalog 自行加入
 *
 * @param {Object|null} ext 輸入擴充 { layers, arms }
 * @param {Array|Object|null} catalog 輸入基底目錄
 * @returns {Array|null} 回傳名之陣列(去重);無擴充 ⇒ null
 * @example
 * let base = { arms: [{ id: 'px', realize: [{ kind: 'operator', op: 'P.x' }] }, { id: 'py', realize: [{ kind: 'operator', op: 'P.y' }] }] }
 * let ext = { arms: [{ op: 'revise', id: 'px', patch: { realize: [{ kind: 'operator', op: 'P.x2' }] } }, { op: 'add', arm: { id: 'kz', realize: [{ kind: 'carrier', name: 'far' }] } }] }
 * console.log(lenientNamesOf(ext, base).join(','), lenientNamesOf(null, base))
 * // => P.x,P.x2,far null
 */
export let lenientNamesOf = (ext, catalog) => {
    if (!ext) { return null }
    let ids = new Set(lenientIdsOf(ext))
    let names = []
    let push = (realize) => { names.push(...realizeNamesOf(realize)) }
    for (let a of catalogArmsOf(catalog)) { if (isPlainObject(a) && ids.has(a.id)) { push(a.realize) } }
    for (let e of isArray(ext.arms) ? ext.arms : []) {
        if (isPlainObject(e) && e.op === 'add') { push(get(e, 'arm.realize')) }
        if (isPlainObject(e) && e.op === 'revise') { push(get(e, 'patch.realize')) }
    }
    return [...new Set(names)]
}

/**
 * 讀狀態根之擴充並套到基底目錄上(供使用端於組裝領域之前取得「生效之目錄」—— 如由目錄推導算子與載體者);有擴充時另做不對登錄表之結構檢查(validateCatalog),不合即拋;無擴充檔 ⇒ 原樣回傳基底
 *
 * @param {Object} paths 輸入 makePaths 之回傳(取 govExtensions)
 * @param {Array|Object|null} catalog 輸入基底目錄
 * @returns {Array|Object|null} 回傳生效之目錄
 * @throws {Error} 擴充檔損壞、擴充之結構不合或套用後之目錄結構不合時拋出
 * @example
 * import fs from 'fs'
 * import os from 'os'
 * import path from 'path'
 * import { makePaths } from 'w-recursi/src/core/paths.mjs'
 *
 * let dir = fs.mkdtempSync(path.join(os.tmpdir(), 'w-recursi-'))
 * let base = { layers: { P: '位置層' }, arms: [{ id: 'px', layer: 'P', title: 'x 步進', status: 'sparse', realize: [{ kind: 'manual', how: '—' }] }] }
 * console.log(effectiveCatalogOf(makePaths(dir), base) === base)
 * // => true
 * fs.rmSync(dir, { recursive: true, force: true })
 */
export let effectiveCatalogOf = (paths, catalog) => {
    let ext = readArmExtensions(paths)
    let out = applyArmExtensions(catalog, ext)
    if (out !== catalog) { let c = validateCatalog(out); if (c.errors.length) { throw new Error(`臂目錄之擴充:套用後之目錄不合:\n  ${c.errors.join('\n  ')}`) } }
    return out
}

/**
 * 新臂之 id:層代號 + 全目錄之下一個序號(至少 2 位)+ 底線 + 英數名;與目錄內既有之 id 相同者序號再遞增(2026-11-14 外審 Argon #7:與明給之 arm.id 混用時曾配出既有之 id)
 *
 * @param {Array|Object|null} catalog 輸入目錄
 * @param {String} layer 輸入層代號
 * @param {String} name 輸入英數名(非英數字元略去)
 * @returns {String} 回傳臂 id
 * @example
 * console.log(nextArmId({ arms: [{ id: 'A01_a', n: 1 }, { id: 'B02_b', n: 2 }] }, 'K', 'entry style'))
 * // => K03_entrystyle
 *
 * console.log(nextArmId({ arms: [{ id: 'A01_a', n: 1 }, { id: 'A03_dup', n: 2 }] }, 'A', 'dup'))
 * // => A04_dup
 */
export let nextArmId = (catalog, layer, name) => {
    let arms = catalogArmsOf(catalog)
    let ids = new Set(arms.map((a) => (isPlainObject(a) ? a.id : null)))
    let n = arms.reduce((m, a) => Math.max(m, Number.isInteger(get(a, 'n')) ? a.n : 0), 0) || arms.length
    let idOf = (k) => `${layer}${String(k).padStart(2, '0')}_${String(name).replace(/[^A-Za-z0-9]/g, '')}`
    let k = n + 1
    while (ids.has(idOf(k))) { k++ }
    return idOf(k)
}
/**
 * 新層之代號:尚未使用之下一個大寫字母
 *
 * @param {Array|Object|null} catalog 輸入目錄
 * @returns {String} 回傳層代號
 * @throws {Error} 26 個字母皆已使用時拋出
 * @example
 * console.log(nextLayerId({ layers: { A: 'a', B: 'b', D: 'd' }, arms: [] }), nextLayerId(null))
 * // => C A
 */
export let nextLayerId = (catalog) => {
    let layers = catalogLayersOf(catalog)
    for (let c = 65; c <= 90; c++) { let k = String.fromCharCode(c); if (layers[k] === undefined) { return k } }
    throw new Error('nextLayerId: 層之代號已用完')
}

let lineOf = (x) => `〔${KIND_LABEL[x.kind] || x.kind}〕${x.arm ? `${x.arm} ` : ''}${x.title || ''}${x.layer ? `(層:${x.layer})` : ''}:${x.mechanism}${x.boundary ? `;與既有臂之界線:${x.boundary}` : ''}${x.evidence ? `(依據:${x.evidence})` : ''}${isArray(x.sources) && x.sources.length ? `(出處:${x.sources.join('、')})` : ''}`
let whoOf = (x) => `${isNil(x.round) ? '' : `第 ${x.round} 輪;`}${x.from || '未註明'}`
/**
 * 提案帳之渲染(markdown;純函式):待審在前(依編號)、已審定在後(依編號新者在前)
 *
 * @param {Object} ledger 輸入提案帳 { seq, items }
 * @returns {String} 回傳 markdown
 * @example
 * let md = renderProposals({ seq: 1, items: [{ id: 'P001', round: 3, from: 'dream', kind: 'new', layer: 'P', title: 'z 步進', mechanism: '對 z 軸步進', status: 'pending', decision: null }] })
 * console.log(md.split('\n').filter((l) => /^## |^- /.test(l)).join(' | '))
 * // => ## 待審(1 筆) | - **P001**(第 3 輪;dream)〔新臂〕z 步進(層:P):對 z 軸步進 | ## 已審定(0 筆;新者在前)
 */
export let renderProposals = (ledger) => {
    let items = isArray(get(ledger, 'items')) ? ledger.items : []
    let pend = items.filter((x) => x.status === 'pending')
    let done = items.filter((x) => x.status !== 'pending').reverse()
    let L = ['# 治理:提案帳', '', '> 由引擎產生,不手改(正本 proposals.json)。下層提案、主控審定;通過之臂目錄變更寫入 extensions.json,下一次組裝(或 reloadArms)起生效。', '']
    L.push(`## 待審(${pend.length} 筆)`, '', pend.length ? pend.map((x) => `- **${x.id}**(${whoOf(x)})${lineOf(x)}${x.note ? `〔${x.note}〕` : ''}`).join('\n') : '(無)', '')
    L.push(`## 已審定(${done.length} 筆;新者在前)`, '', done.length ? done.map((x) => `- **${x.id}** 〔${STATUS_LABEL[x.status] || x.status}${get(x, 'decision.ref') ? ` ⇒ \`${x.decision.ref}\`` : ''}〕(${whoOf(x)};${KIND_LABEL[x.kind] || x.kind})${x.arm ? `${x.arm} ` : ''}${x.title || ''}:${String(x.mechanism).slice(0, 160)} —— 審定(${get(x, 'decision.by', '-')}):${get(x, 'decision.why', '')}`).join('\n') : '(無)', '')
    return L.join('\n')
}

/**
 * 治理工廠(createRsi 以之組 rsi.governance):提案帳、審定、擴充、起跑之把關、收件匣。寫入皆於 governance 鎖內;通過之臂目錄變更先以「基底 + 新擴充」試建目錄(結構檢查與 cfg.arms.use 之核對),不合 ⇒ 拋錯且不寫任何檔;提案帳存在而損壞 ⇒ propose / decide / list 拋錯(不以新帳覆寫)。
 * 審定之提交點為擴充檔之原子寫;其後寫提案帳(失敗 ⇒ 拋 code RSI_GOVERNANCE_PARTIAL,重跑同一審定即完成)、重載本程序之目錄(失敗 ⇒ 回 reloadError)、重產 PROPOSALS.md(失敗 ⇒ 回 renderError);前次審定中斷而擴充檔已有本提案之項者,重審之 accept 只補記提案帳(recovered),改判不採 ⇒ 拋 code RSI_GOVERNANCE_COMMITTED
 *
 * @param {Object} rsi 輸入實例(取 paths、cfg、baseCatalog、checkCatalog、reloadArms、arms)
 * @returns {Object} 回傳 { opt(生效之治理選項), ledger(), list(opt), pending(), propose(o), decide(id, o), extensions(), effectiveCatalog(), gate(o), inbox(), render(), exists() };propose 回 { added, repeated, invalid, renderError? },decide 回 { item, extension, reloaded, reloadError?, renderError?, recovered? }
 * @example
 * import fs from 'fs'
 * import os from 'os'
 * import path from 'path'
 * import { createRsi } from 'w-recursi/src/index.mjs'
 *
 * let dir = fs.mkdtempSync(path.join(os.tmpdir(), 'w-recursi-'))
 * let domain = {
 *     name: 'toy',
 *     score: (raw) => ({ S: raw.v, ladder: raw.v >= 2 ? 1 : 0, flags: {} }),
 *     validateSpec: (s) => { if (typeof s.x !== 'number') { throw new Error('x 須為數字') } },
 *     operators: [{ name: 'inc', family: 'f', applies: () => true, children: (s) => [{ spec: { x: s.x + 1 }, params: { by: 1 } }] }],
 *     carriers: { root: { x: 0 } },
 *     evaluator: { mode: 'inProcess', evaluate: async (spec) => ({ v: spec.x }) },
 * }
 * let catalog = { layers: { L: '層' }, arms: [{ id: 'A1', layer: 'L', title: '遞增', status: 'sparse', realize: [{ kind: 'operator', op: 'inc' }] }] }
 * let rsi = createRsi({ root: dir, domain, arms: { catalog }, governance: { requireReview: true } }).init()
 * let p = rsi.governance.propose({ round: 1, from: 'dream', items: [{ kind: 'new', layer: 'L', title: '遞增(手動)', mechanism: '以另一種方式遞增' }] })
 * console.log(p.added.map((x) => x.id).join(','), rsi.governance.gate({ round: 2 }).pending.length)
 * // => P001 1
 * let d = rsi.governance.decide('P001', { accept: true, why: '機制獨立', arm: { id: 'A2', layer: 'L', title: '遞增(手動)', realize: [{ kind: 'manual', how: '待實作' }] } })
 * console.log(d.item.status, rsi.arms.size, rsi.governance.gate({ round: 2 }))
 * // => accepted 2 null
 * fs.rmSync(dir, { recursive: true, force: true })
 */
export let makeGovernance = (rsi) => {
    let { paths } = rsi
    let gc = isPlainObject(get(rsi.cfg, 'governance')) ? rsi.cfg.governance : {}
    let G = Object.freeze({ ...GOVERNANCE_DEFAULTS, ...omitBy(gc, isNil) })
    let lockOpt = { dir: paths.governance, name: 'governance' }
    //提案帳:不存在 ⇒ 空帳;存在而非 JSON 或形不合 ⇒ 拋錯(不當成空帳, 免被下一次寫入覆蓋)
    //  形:{ seq:非負安全整數, items:[] };每列 id 為 P + 數字且唯一、seq ≥ 最大編號(免重用編號)、kind / status 在值域、mechanism 為字串、已審定者附 decision 物件
    let ledger = () => {
        if (!fs.existsSync(paths.govProposals)) { return { seq: 0, items: [] } }
        let l = null
        try { l = readJson(paths.govProposals, null) }
        catch (e) { throw new Error(`治理:提案帳非 JSON(${paths.govProposals})`) }
        let bad = (why) => new Error(`治理:提案帳須為 { seq, items:[] }(${why};${paths.govProposals})`)
        if (!isPlainObject(l) || !isArray(l.items)) { throw bad('頂層') }
        if (!Number.isSafeInteger(l.seq) || l.seq < 0) { throw bad('seq 須為非負整數') }
        let ids = new Set()
        for (let [i, x] of l.items.entries()) {
            let m = isPlainObject(x) && isString(x.id) ? /^P(\d+)$/.exec(x.id) : null
            if (!m || ids.has(x.id)) { throw bad(`items[${i}] 之 id 不合式或重複`) }
            ids.add(x.id)
            if (Number(m[1]) > l.seq) { throw bad(`items[${i}] 之編號大於 seq`) }
            if (!PROPOSAL_KINDS.includes(x.kind) || !PROPOSAL_STATUS.includes(x.status) || !isString(x.mechanism)) { throw bad(`items[${i}] 之 kind / status / mechanism`) }
            let d = x.decision
            if (x.status === 'pending' ? !isNil(d) : !(isPlainObject(d) && str(d.at) && str(d.by) && str(d.why))) { throw bad(`items[${i}] 之 decision 與 status 不一致或缺 at / by / why`) }
        }
        return l
    }
    //擴充:讀檔(損壞即拋)並以「基底 + 擴充」做不對登錄表之結構檢查(深層損壞亦拋;status / 報告 / 生效目錄 / 審定一致)
    let extensions = () => {
        let ext = readArmExtensions(paths) || { layers: [], arms: [] }
        if (ext.layers.length || ext.arms.length) { let c = validateCatalog(applyArmExtensions(rsi.baseCatalog, ext)); if (c.errors.length) { throw new Error(`臂目錄之擴充:套用後之目錄不合:\n  ${c.errors.join('\n  ')}`) } }
        return ext
    }
    //寫提案帳(正本;失敗即拋);PROPOSALS.md 為衍生檔, 重產失敗不使呼叫失敗(回錯誤字串)
    let saveLedger = (led) => { assertWriteRoot(paths.govProposals, { what: 'governance(proposals.json)' }); writeJson(paths.govProposals, led) }
    let renderMd = (led) => { try { writeFileAtomic(paths.govProposalsMd, renderProposals(led)); return null } catch (e) { return String((e && e.message) || e).split('\n')[0].slice(0, 300) } }
    let errLine = (e) => String((e && e.message) || e).split('\n')[0].slice(0, 300)
    let list = ({ status = null, kind = null } = {}) => ledger().items.filter((x) => (status === null || x.status === status) && (kind === null || x.kind === kind))
    let pending = () => list({ status: 'pending' })
    //同鍵:類別 + 臂 + 層 + 名稱(空白與大小寫不計);revise 另含修訂內容(同一臂之不同修訂為不同提案)
    let keyOf = (p) => [p.kind, norm(p.arm), norm(p.layer), norm(p.title), p.kind === 'revise' ? norm(p.mechanism) : ''].join('|')
    //提案:items 不合格者不入帳(列於 invalid);與帳上待審者同鍵者不另立(repeated);round = 提出之輪號、from = 提出者(層名或節點 id;自由字串)
    //  items 未給或 null ⇒ 空;給了而非陣列 ⇒ 拋錯(不與合法之空陣列混同)
    let propose = ({ round = null, from = null, items = null } = {}) => {
        if (!isNil(items) && !isArray(items)) { throw new Error('governance.propose: items 須為提案陣列') }
        return withLock(() => proposeIn({ round, from, items: items || [] }), lockOpt)
    }
    let proposeIn = ({ round, from, items }) => {
        let led = ledger()
        let armIds = rsi.arms ? rsi.arms.arms.map((a) => a.id) : []
        let added = [], repeated = [], invalid = []
        for (let [i, p] of items.entries()) {
            let errs = checkProposals([p]).errors
            if (errs.length) { invalid.push({ index: i, why: errs[0].replace(/^提案\[0\]:/, '') }); continue }
            let old = led.items.find((x) => x.status === 'pending' && keyOf(x) === keyOf(p))
            if (old) {
                let extra = [str(p.mechanism) !== old.mechanism ? str(p.mechanism) : null, str(p.evidence)].filter(Boolean).join(';')
                old.note = `${old.note ? `${old.note};` : ''}${Number.isInteger(round) ? `第 ${round} 輪` : ''}再提(${str(from) || '未註明'})${extra ? `:${extra.slice(0, 300)}` : ''}`
                repeated.push(old)
                continue
            }
            if (!Number.isSafeInteger(led.seq + 1)) { throw new Error('governance.propose: 提案編號已達上限(seq)') }
            led.seq = led.seq + 1
            let it = {
                id: `P${String(led.seq).padStart(3, '0')}`, at: nowIso(), round: Number.isInteger(round) ? round : null, from: str(from), kind: p.kind,
                arm: str(p.arm), layer: str(p.layer), title: str(p.title), mechanism: str(p.mechanism), boundary: str(p.boundary), evidence: str(p.evidence), sources: isArray(p.sources) ? p.sources.slice() : [],
                note: ['revise', 'retire'].includes(p.kind) && !armIds.includes(str(p.arm)) ? '所指之臂不在目錄' : null,
                status: 'pending', decision: null,
            }
            led.items.push(it)
            added.push(it)
        }
        let renderError = null
        if (added.length || repeated.length) { saveLedger(led); renderError = renderMd(led) }
        return { added, repeated, invalid, ...(renderError ? { renderError } : {}) }
    }
    //審定:accept(true 通過 / false 不採)、why(必填)、by(審定者)
    //  通過之 new:arm = 目錄條目(須含 id 與 realize;layer / title 未給時取提案者);layer:layer = { id, title }(未給 ⇒ id 取下一個字母、title 取提案者)
    //  revise:arm = 要改之欄(patch;id 可省, 給則須同所指之臂;值為 null 之鍵 = 刪除該鍵, undefined 之鍵視同未給);retire:無須另給;note:只記審定
    //  name(通過新臂且 arm.id 未給時):英數名 ⇒ 於鎖內依最新之生效目錄以 nextArmId 配號(免兩個審定者於鎖外配到同號)
    //  realize 之字串欄內之 {id}(new / revise)⇒ 代換為該臂之 id(name 配號者為鎖內配出之 id;2026-11-14,Gitea #144)
    //  寫入擴充檔之項須為純資料(JSON 往返不變;函數、Date、非有限數等 ⇒ 拒, 免本程序之目錄與檔內者不一致)
    //  回 { item, extension(擴充檔之該筆 | null), reloaded(本程序之 rsi.arms 是否已重載), reloadError?, renderError?, recovered?(補記前次中斷之審定) }
    let decide = (id, { accept, why, by = 'controller', arm = null, layer = null, name = null } = {}) => withLock(() => {
        if (!isBoolean(accept) || !str(why)) { throw new Error('governance.decide: 須給 accept(布林)與 why(審定理由)') }
        let led = ledger()
        let it = led.items.find((x) => x.id === id)
        if (!it) { throw new Error(`governance.decide: 提案帳無此筆 ${id}`) }
        if (it.status !== 'pending') { throw new Error(`governance.decide: 提案 ${id} 已審定(${it.status})`) }
        let at = nowIso()
        let byS = str(by) || 'controller'
        //origin 記完整之審定(誰、何時、理由):擴充檔即提交點, 中斷後之補記據此還原原決定
        let origin = { proposal: it.id, round: it.round, from: it.from, by: byS, at, why: str(why) }
        let ext = extensions()
        //本次審定之擴充項(不含 origin;exts = 配號所據之擴充;forceId = 補記時沿用已生效之 id, 不重新配號)
        let build = (exts, forceId = null) => {
            if (it.kind === 'new') {
                let aid = isPlainObject(arm) ? str(arm.id) : null
                let lyr = (isPlainObject(arm) && str(arm.layer)) || it.layer
                if (!aid && forceId && str(name)) { aid = forceId }
                if (!aid && str(name) && lyr) { aid = nextArmId(applyArmExtensions(rsi.baseCatalog, exts), lyr, name) }
                if (!isPlainObject(arm) || !aid) { throw new Error('governance.decide: 通過新臂須給 arm(目錄條目;至少 realize,id 或 name 擇一)') }
                //值為 undefined 之鍵視同未給(層 / 名稱取提案者;2026-11-14 外審 Argon #5)
                let a = omitBy({ layer: lyr, title: it.title, ...omitBy(omit(arm, ['origin', 'n']), isUndefined), id: aid }, isNil)
                if (a.realize !== undefined) { a.realize = fillId(a.realize, aid) }
                return { op: 'add', arm: a }
            }
            if (it.kind === 'layer') { return { id: str(get(layer, 'id')) || forceId || nextLayerId(applyArmExtensions(rsi.baseCatalog, exts)), title: str(get(layer, 'title')) || it.title } }
            if (it.kind === 'revise') {
                if (isPlainObject(arm) && !isNil(arm.id) && arm.id !== it.arm) { throw new Error(`governance.decide: 修訂之 arm.id 須同所指之臂 ${it.arm}`) }
                let p = isPlainObject(arm) ? omitBy(omit(arm, ['id']), isUndefined) : {}
                if (!isNil(p.realize)) { p.realize = fillId(p.realize, it.arm) }
                return { op: 'revise', id: it.arm, patch: p }
            }
            return { op: 'retire', id: it.arm, why: str(why) }
        }
        let refOf = (e) => (e.op === 'add' ? get(e, 'arm.id', null) : e.id)
        //前次審定已寫入擴充檔(提交點)而未寫提案帳:該項已生效、後續審定可能已依賴之 ⇒ accept 只補記提案帳(不重寫、不搬動;審定者 / 時間 / 理由取原決定), 不採則拒;
        //  本次另給了 arm / layer / name 而與已生效之內容不同 ⇒ 拒(不把不同之請求當成原決定)
        let mine = (x) => get(x, 'origin.proposal') === it.id
        let prior = [...ext.layers.filter(mine), ...ext.arms.filter(mine)]
        let entry = null, ref = null, recovered = null
        if (prior.length) {
            let committed = Object.assign(new Error(`governance.decide: 提案 ${id} 之通過已寫入擴充檔而生效(前次審定於寫提案帳前中斷);只能以 accept 補記原決定(不另給不同之 arm / layer / name),或另提 retire / revise 撤回`), { code: 'RSI_GOVERNANCE_COMMITTED' })
            if (!accept) { throw committed }
            entry = prior[0]
            if (it.kind !== 'retire' && (!isNil(arm) || !isNil(layer) || !isNil(name))) {
                //比對時沿用已生效之 id(由引擎配出者不重配:其後之審定會改變配號所據之目錄;2026-11-14 外審 Argon #6)
                let want = null
                try { want = build(null, refOf(entry)) } catch (e) { throw committed }
                if (!isEqual(want, omit(entry, ['origin']))) { throw committed }
            }
            ref = refOf(entry)
            recovered = { at, by: byS, ...(str(why) ? { why: str(why) } : {}) }
        }
        let prevCatalog = null
        if (!prior.length && accept && VOCAB_KINDS.includes(it.kind)) {
            entry = { ...build(ext), origin }
            if (!isPureData(entry)) { throw new Error('governance.decide: arm / layer 須為純資料(JSON 可表示且往返不變;不得含函數、Date、非有限數等)') }
            prevCatalog = applyArmExtensions(rsi.baseCatalog, ext)
            if (it.kind === 'layer') { ext.layers.push(entry) }
            else { ext.arms.push(entry) }
            ref = refOf(entry)
        }
        if (entry && !recovered) {
            //試建目錄 = 組裝之同一檢查(createRsi / reloadArms 亦為 makeArms(基底 + 擴充, lenientIdsOf(擴充))+ cfg.arms.use 之核對):結構檢查、以本程序之登錄表 / known 組一次
            //  (lenient 之臂之落成名未登錄者降為不可選, 見 arms.mjs)、核對 cfg.arms.use;不合 ⇒ 拋錯, 以下皆不執行(兩檔不變)
            //  下一次組裝之登錄表若由目錄推導(README §2c), 只有 lenient 之臂之登錄會隨本次審定而變 —— 其未登錄不使組裝失敗(2026-11-14,Gitea #144:停用之臂原不在 lenient ⇒ 試建以本程序之登錄表通過、下一次組裝拋錯)
            let nextCatalog = applyArmExtensions(rsi.baseCatalog, ext)
            rsi.checkCatalog(nextCatalog, { lenient: lenientIdsOf(ext), lenientNames: lenientNamesOf(ext, rsi.baseCatalog) })
            //預設選臂(cfg.arms.use)之落成名若因本次審定而登錄與否可能改變(本次觸及之臂、或與之共用落成名)⇒ 試建之登錄表(本程序組裝時者)不能代表下一次組裝 ⇒ 收嚴:拒
            //  (由生效目錄推導登錄之領域, 下一次組裝該臂即可能未登錄而不可選, 組裝於 cfg.arms.use 之核對拋錯;2026-11-14 外審 Argon #3)
            let use = get(rsi.cfg, 'arms.use')
            //  只看停用 / 修訂之臂(新增之臂與新落成名不會使既有之登錄消失);名取該臂於本次之前之落成名(可能不再登錄者;外審 hydrogen 第二輪 #3:原含新增之臂與新名而過嚴)
            if (isArray(use) && use.length && entry.op !== 'add' && it.kind !== 'layer') {
                let ids = new Set(lenientIdsOf({ layers: [], arms: [entry] }))
                let prevById = Object.fromEntries(catalogArmsOf(prevCatalog).map((a) => [a.id, a]))
                let names = new Set([...ids].flatMap((x) => realizeNamesOf(get(prevById, [x, 'realize']))))
                let byId = Object.fromEntries(catalogArmsOf(nextCatalog).map((a) => [a.id, a]))
                let risky = use.filter((u) => byId[u] && (ids.has(u) || realizeNamesOf(byId[u].realize).some((n) => names.has(n))))
                if (risky.length) { throw new Error(`臂目錄與 cfg.arms.use 被拒:${risky.join('、')}(其落成名之登錄可能隨本審定而變 —— 由生效目錄推導登錄之領域, 下一次組裝即不可選而拋錯;先自 cfg.arms.use 移除該臂再審定)`) }
            }
            //提交點:擴充檔之原子寫
            assertWriteRoot(paths.govExtensions, { what: 'governance(extensions.json)' })
            writeJson(paths.govExtensions, ext)
        }
        it.status = accept ? 'accepted' : 'rejected'
        //補記:決定取擴充檔所記之原決定(origin 無 why 者以本次理由補), 本次呼叫記於 recovered
        let po = recovered ? get(entry, 'origin', {}) || {} : null
        it.decision = recovered
            ? { at: str(po.at) || at, by: str(po.by) || byS, why: str(po.why) || str(why), ...(ref ? { ref } : {}), recovered }
            : { at, by: byS, why: str(why), ...(ref ? { ref } : {}) }
        try { saveLedger(led) }
        catch (e) {
            if (!entry || recovered) { throw e }
            throw Object.assign(new Error(`governance.decide: 提案 ${id} 之擴充已寫入而生效,但提案帳未能寫入(${errLine(e)});重跑同一審定即補記`), { code: 'RSI_GOVERNANCE_PARTIAL', cause: e })
        }
        let out = { item: it, extension: entry, reloaded: false, ...(recovered ? { recovered: true } : {}) }
        if (entry) { try { rsi.reloadArms(); out.reloaded = true } catch (e) { out.reloadError = errLine(e) } }
        let renderError = renderMd(led)
        if (renderError) { out.renderError = renderError }
        return out
    }, lockOpt)
    let effectiveCatalog = () => applyArmExtensions(rsi.baseCatalog, extensions())
    //起跑之把關:requireReview 且有逾期之待審(類別 ∈ gateKinds;提出之輪 + maxPendingRounds < 將起跑之輪;輪號不明者一律視為逾期)⇒ { why, pending };否則 null
    let gate = ({ round = null } = {}) => {
        if (G.requireReview !== true) { return null }
        let over = pending().filter((x) => G.gateKinds.includes(x.kind) && (!Number.isInteger(x.round) || !Number.isInteger(round) || x.round + G.maxPendingRounds < round))
        return over.length ? { why: `待審之提案 ${over.length} 筆(${over.map((x) => x.id).join(', ')})已逾 ${G.maxPendingRounds} 輪未審:主控審定後再起跑(rsi.governance.decide;或以 ackPending 明示越過)`, pending: over } : null
    }
    //收件匣:主控之單一入口 —— 待審之提案 + 引擎既有之兩種請求標記(標記非 JSON ⇒ 視同無, 同 readGenRequest)
    let inbox = () => {
        let ps = pending()
        let generation = readJsonSafe(paths.genRequest), policyDev = readJsonSafe(paths.devRequest)
        return { proposals: ps, byKind: countBy(ps, 'kind'), generation, policyDev, n: ps.length + (generation ? 1 : 0) + (policyDev ? 1 : 0) }
    }
    let render = () => renderProposals(ledger())
    //opt = 生效之治理選項(GOVERNANCE_DEFAULTS 併 cfg.governance 之非空鍵;凍結)
    return { opt: G, ledger, list, pending, propose, decide, extensions, effectiveCatalog, gate, inbox, render, exists: () => fs.existsSync(paths.govProposals) || fs.existsSync(paths.govExtensions) }
}
